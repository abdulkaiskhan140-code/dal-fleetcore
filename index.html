📖 Firebase setup guide (3 minutes) — tap to open
Step 1: Firebase project banao
1. console.firebase.google.com kholo
2. Add project → naam do (e.g. "dal-fleetcore") → Continue
3. Analytics: Disable (zaroori nahi) → Create project
Step 2: Firestore Database banao
1. Left menu se Build → Firestore Database kholo
2. Create database → Location: asia-south1 (Mumbai, closest to Saudi)
3. Mode: Start in test mode → Enable
Step 3: Web app register karo
1. Project Overview pe ⚙️ icon → Project settings
2. Scroll niche → Your apps → </> (Web) icon click karo
3. Nickname: "DAL FleetCore" → Register app
4. Aage screen pe firebaseConfig object dikhega — uske values copy karke yahaan paste karo:
apiKey, authDomain, projectId, storageBucket, appId
Step 4: Security Rules (1 minute)
1. Firestore Database → Rules tab
2. Yeh rules paste karo (sirf 30 din tak open access):
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /entries/{doc} {
      allow read, write: if true;
    }
  }
}
3. Publish click karo
⚠️ Production ke liye baad mein authentication enable karna behtar hai.
Step 5: Connect
Upar 5 fields fill karo → Connect Firebase dabao → done! ☁️

function doPost(e) { try { // Accept both application/json and text/plain (no-cors sends text/plain) var raw = e.postData ? e.postData.contents : ''; var d = JSON.parse(raw); var ss = SpreadsheetApp.getActiveSpreadsheet(); // REPLACE action — Admin panel fleet sync if (d.action === 'replace') { var sh = ss.getSheetByName(d.sheet); if (!sh) sh = ss.insertSheet(d.sheet); sh.clearContents(); if (SHEET_HEADERS[d.sheet]) { sh.appendRow(SHEET_HEADERS[d.sheet]); sh.getRange(1,1,1,SHEET_HEADERS[d.sheet].length).setBackground('#1a73e8').setFontColor('#fff').setFontWeight('bold'); sh.setFrozenRows(1); } if (d.rows && d.rows.length) d.rows.forEach(function(r){ sh.appendRow(r); }); return out({status:'success', msg: d.sheet + ' updated'}); } // APPEND action — Normal form submission var shName = d.sheet; if (!shName) return out({status:'error', msg:'No sheet name'}); var sh = ss.getSheetByName(shName); if (!sh) { sh = ss.insertSheet(shName); if (SHEET_HEADERS[shName]) { sh.appendRow(SHEET_HEADERS[shName]); sh.getRange(1,1,1,SHEET_HEADERS[shName].length).setBackground('#1a73e8').setFontColor('#fff').setFontWeight('bold'); sh.setFrozenRows(1); } } var ts = Utilities.formatDate(new Date(), 'Asia/Riyadh', 'dd/MM/yyyy HH:mm:ss'); var row = d.row || []; row.unshift(ts); sh.appendRow(row); return out({status:'success'}); } catch(err) { return out({status:'error', msg: err.toString()}); } } function doGet(e) { try { var action = (e && e.parameter && e.parameter.action) || 'ping'; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'fleet') { var fleet = {}; var fs = ss.getSheetByName('FLEET_DATA'); if (fs && fs.getLastRow() > 1) { var rows = fs.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { var p = rows[i][0]; if (!p) continue; if (!fleet[p]) fleet[p] = []; fleet[p].push({plate:rows[i][1]||'',trailer:rows[i][2]||'',driver:rows[i][3]||'',make:rows[i][4]||'',type:rows[i][5]||''}); } } var trailers = []; var ts = ss.getSheetByName('TRAILERS'); if (ts && ts.getLastRow() > 1) { var tr = ts.getDataRange().getValues(); for (var j = 1; j < tr.length; j++) { if (tr[j][0]) trailers.push(tr[j][0]); } } return out({status:'ok', fleet:fleet, trailers:trailers}); } return out({status:'ok', msg:'DAL FleetCore API v3 ready ✅'}); } catch(err) { return out({status:'error', msg: err.toString()}); } } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Run once manually to create all sheets function initAllSheets() { var ss = SpreadsheetApp.getActiveSpreadsheet(); Object.keys(SHEET_HEADERS).forEach(function(name) { var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(SHEET_HEADERS[name]); sh.getRange(1,1,1,SHEET_HEADERS[name].length).setBackground('#1a73e8').setFontColor('#fff').setFontWeight('bold'); sh.setFrozenRows(1); } }); Browser.msgBox('✅ Done! All sheets created.'); }
